You are a Principal Software Architect and Senior Full-Stack Engineer. Your objective is to build a production-ready, ultra-lightweight, single-binary OpenID Connect (OIDC) Identity Provider named **OneAuth**.

1. **Language & Framework**: Go 1.22+ using standard library or lightweight routers (e.g., standard `net/http` with modern ServeMux or `chi`).
2. **Storage Architecture**:
   - **Persistent Config**: Pure SQLite with WAL mode (`modernc.org/sqlite`). NO external database (MySQL/PostgreSQL) allowed.

   - Built-in lightweight Admin Console (Client management, UI customizer, Bot settings, Audit logs).
4. **Protocol Compliance**:
   - Strictly conforms to OpenID Connect Core 1.0 and RFC 6749 (Authorization Code Flow with PKCE support).
   - Generates and self-manages a persistent RSA 2048-bit keypair on the first run, exposing standard `/.well-known/jwks.json`.
5. **Protocol Adapter**:
   - Implements an RFC 6455 compliant WebSocket server at `/ws/onebot` receiving OneBot v11 and v12 Reverse WebSocket events.

	var err error

	return DB, nil
}

6.3 内存状态机与原子核销 (internal/session/manager.go)Gopackage session

import (
	"crypto/rand"
	"math/big"
	"sync"
	"time"
)

type SessionStatus string

const (
	StatusPending  SessionStatus = "PENDING"
	StatusVerified SessionStatus = "VERIFIED"
	StatusConsumed SessionStatus = "CONSUMED"
)

type AuthSession struct {
	SessionID     string
	ClientID      string
	RedirectURI   string
	State         string
	CodeChallenge string
	VerifyCode    string
	QQNumber      string
	AuthCode      string
	Status        SessionStatus
	ExpiresAt     time.Time
	NotifyChan    chan struct{}
}

type Manager struct {
	mu            sync.RWMutex
	sessions      map[string]*AuthSession
	codeIndex     map[string]*AuthSession
	authCodeIndex map[string]*AuthSession
}

var DefaultManager = NewManager()

func NewManager() *Manager {
	m := &Manager{
		sessions:      make(map[string]*AuthSession),
		codeIndex:     make(map[string]*AuthSession),
		authCodeIndex: make(map[string]*AuthSession),
	}
	go m.startCleaner()
	return m
}

func (m *Manager) CreateSession(clientID, redirectURI, state, challenge string, ttlSeconds int) (*AuthSession, error) {
	m.mu.Lock()
	defer m.mu.Unlock()

	code, err := m.generateUniqueCode(6)
	if err != nil {
		return nil, err
	}

	sessionID, _ := generateRandomHex(16)
	sess := &AuthSession{
		SessionID:     sessionID,
		ClientID:      clientID,
		RedirectURI:   redirectURI,
		State:         state,
		CodeChallenge: challenge,
		VerifyCode:    code,
		Status:        StatusPending,
		ExpiresAt:     time.Now().Add(time.Duration(ttlSeconds) * time.Second),
		NotifyChan:    make(chan struct{}),
	}

	m.sessions[sessionID] = sess
	m.codeIndex[code] = sess
	return sess, nil
}

func (m *Manager) VerifyCode(code, qqNumber string) (*AuthSession, bool) {
	m.mu.Lock()
	defer m.mu.Unlock()

	sess, exists := m.codeIndex[code]
	if !exists || time.Now().After(sess.ExpiresAt) || sess.Status != StatusPending {
		return nil, false
	}

	sess.QQNumber = qqNumber
	sess.Status = StatusVerified
	delete(m.codeIndex, code) // 原子核销，防止重放

	close(sess.NotifyChan)
	return sess, true
}

func (m *Manager) IssueAuthCode(sessionID string) (string, *AuthSession, bool) {
	m.mu.Lock()
	defer m.mu.Unlock()

	sess, exists := m.sessions[sessionID]
	if !exists || sess.Status != StatusVerified {
		return "", nil, false
	}

	authCode, _ := generateRandomHex(24)
	sess.AuthCode = authCode
	sess.Status = StatusConsumed
	m.authCodeIndex[authCode] = sess

	return authCode, sess, true
}

func (m *Manager) ExchangeCode(authCode string) (*AuthSession, bool) {
	m.mu.Lock()
	defer m.mu.Unlock()

	sess, exists := m.authCodeIndex[authCode]
	if !exists {
		return nil, false
	}

	delete(m.authCodeIndex, authCode)
	return sess, true
}

func (m *Manager) GetSession(sessionID string) (*AuthSession, bool) {
	m.mu.RLock()
	defer m.mu.RUnlock()
	sess, exists := m.sessions[sessionID]
	return sess, exists
}

func (m *Manager) generateUniqueCode(length int) (string, error) {
	const charset = "23456789ABCDEFGHJKMNPQRSTUVWXYZ"
	for {
		b := make([]byte, length)
		for i := range b {
			idx, err := rand.Int(rand.Reader, big.NewInt(int64(len(charset))))
			if err != nil {
				return "", err
			}
			b[i] = charset[idx.Int64()]
		}
		candidate := string(b)
		if _, exists := m.codeIndex[candidate]; !exists {
			return candidate, nil
		}
	}
}

func (m *Manager) startCleaner() {
	ticker := time.NewTicker(20 * time.Second)
	for range ticker.C {
		now := time.Now()
		m.mu.Lock()
		for sid, s := range m.sessions {
			if now.After(s.ExpiresAt) {
				delete(m.codeIndex, s.VerifyCode)
				delete(m.authCodeIndex, s.AuthCode)
				delete(m.sessions, sid)
			}
		}
		m.mu.Unlock()
	}
}

func generateRandomHex(length int) (string, error) {
	const hexChars = "0123456789abcdef"
	b := make([]byte, length)
	for i := range b {
		idx, _ := rand.Int(rand.Reader, big.NewInt(16))
		b[i] = hexChars[idx.Int64()]
	}
	return string(b), nil
}

func HandleWebSocket(w http.ResponseWriter, r *http.Request) {
	expectedToken := database.GetSetting("onebot_token", "")
	if expectedToken != "" {
		authHeader := r.Header.Get("Authorization")
		queryToken := r.URL.Query().Get("access_token")
		valid := false
		if strings.HasPrefix(authHeader, "Bearer ") && strings.TrimPrefix(authHeader, "Bearer ") == expectedToken {
			valid = true
		} else if queryToken == expectedToken {
			valid = true
		}
		if !valid {
			http.Error(w, "Unauthorized OneBot Token", http.StatusUnauthorized)
			return
		}
	}

	conn, err := upgrader.Upgrade(w, r, nil)
	if err != nil {
		log.Printf("[OneBot] WS 握手失败: %v\n", err)
		return
	}
	defer conn.Close()

	log.Println("[OneBot] 协议端客户端已成功连接反向 WebSocket")

	for {
		_, msgBytes, err := conn.ReadMessage()
		if err != nil {
			log.Printf("[OneBot] 连接断开: %v\n", err)
			break
		}
		go processEvent(msgBytes)
	}
}

func processEvent(data []byte) {
	var ev GenericEvent
	if err := json.Unmarshal(data, &ev); err != nil {
		return
	}

	targetGroup := database.GetSetting("target_group_id", "")
	var currentGroup, senderQQ string

	if ev.PostType == "message" && ev.MessageType == "group" {
		currentGroup = strconv.FormatInt(ev.GroupIDV11, 10)
		senderQQ = strconv.FormatInt(ev.UserIDV11, 10)
	} else if ev.Type == "message" && ev.DetailType == "group" {
		currentGroup = ev.GroupIDV12
		senderQQ = ev.UserIDV12
	} else {
		return
	}

	if targetGroup != "" && currentGroup != targetGroup {
		return
	}

	code := cleanVerificationCode(ev.Message, ev.RawMessage)
	if len(code) != 6 {
		return
	}

	if sess, ok := session.DefaultManager.VerifyCode(code, senderQQ); ok {
		log.Printf("[OneBot] 成功完成核销: QQ=%s, Code=%s, SessionID=%s\n", senderQQ, code, sess.SessionID)
	}
}

func cleanVerificationCode(msgObj interface{}, raw string) string {
	var sb strings.Builder
	if segments, ok := msgObj.([]interface{}); ok {
		for _, seg := range segments {
			if sm, ok := seg.(map[string]interface{}); ok && sm["type"] == "text" {
				if d, ok := sm["data"].(map[string]interface{}); ok {
					if t, ok := d["text"].(string); ok {
						sb.WriteString(t)
					}
				}
			}
		}
	}

	text := sb.String()
	if text == "" {
		re := regexp.MustCompile(`\[CQ:[^\]]+\]`)
		text = re.ReplaceAllString(raw, "")
	}

	text = strings.ToUpper(strings.TrimSpace(text))
	cleanRegex := regexp.MustCompile(`[^2-9A-HJ-NP-Z]`)
	return cleanRegex.ReplaceAllString(text, "")
}

func (s *Server) HandleAuthorize(w http.ResponseWriter, r *http.Request) {
	q := r.URL.Query()
	clientID := q.Get("client_id")
	redirectURI := q.Get("redirect_uri")
	responseType := q.Get("response_type")
	state := q.Get("state")
	challenge := q.Get("code_challenge")

	if clientID == "" || redirectURI == "" || responseType != "code" {
		http.Error(w, "Invalid request parameters", http.StatusBadRequest)
		return
	}

	var dbRedirects string
	err := database.DB.QueryRow("SELECT redirect_uris FROM oidc_clients WHERE client_id = ?", clientID).Scan(&dbRedirects)
	if err != nil {
		http.Error(w, "Unauthorized client_id", http.StatusUnauthorized)
		return
	}

	var allowedURIs []string
	_ = json.Unmarshal([]byte(dbRedirects), &allowedURIs)
	matched := false
	for _, u := range allowedURIs {
		if u == redirectURI {
			matched = true
			break
		}
	}
	if !matched {
		http.Error(w, "Mismatching redirect_uri", http.StatusBadRequest)
		return
	}

	ttl, _ := strconv.Atoi(database.GetSetting("code_ttl", "180"))
	sess, err := session.DefaultManager.CreateSession(clientID, redirectURI, state, challenge, ttl)
	if err != nil {
		http.Error(w, "Failed to initialize session", http.StatusInternalServerError)
		return
	}

	http.Redirect(w, r, "/login?session_id="+sess.SessionID, http.StatusFound)
}

func (s *Server) HandleToken(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
		return
	}

	_ = r.ParseForm()
	code := r.Form.Get("code")
	clientID := r.Form.Get("client_id")

	sess, ok := session.DefaultManager.ExchangeCode(code)
	if !ok || sess.ClientID != clientID {
		http.Error(w, `{"error":"invalid_grant"}`, http.StatusBadRequest)
		return
	}

	now := time.Now()
	claims := jwt.MapClaims{
		"iss":                s.IssuerURL,
		"sub":                sess.QQNumber,
		"aud":                clientID,
		"exp":                now.Add(24 * time.Hour).Unix(),
		"iat":                now.Unix(),
		"auth_time":          now.Unix(),
		"preferred_username": sess.QQNumber,
		"name":               "QQ用户_" + sess.QQNumber,
		"email":              fmt.Sprintf("%s@qq.com", sess.QQNumber),
		"email_verified":     true,
		"picture":            fmt.Sprintf("https://q1.qlogo.cn/g?b=qq&nk=%s&s=640", sess.QQNumber),
		"qq":                 sess.QQNumber,
	}

	token := jwt.NewWithClaims(jwt.SigningMethodRS256, claims)
	token.Header["kid"] = s.KeyID

	signedIDToken, err := token.SignedString(s.PrivateKey)
	if err != nil {
		http.Error(w, `{"error":"server_error"}`, http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	_ = json.NewEncoder(w).Encode(map[string]interface{}{
		"access_token": signedIDToken,
		"token_type":   "Bearer",
		"expires_in":   86400,
		"id_token":     signedIDToken,
	})
}

func (s *Server) HandleUserInfo(w http.ResponseWriter, r *http.Request) {
	authHeader := r.Header.Get("Authorization")
	if !strings.HasPrefix(authHeader, "Bearer ") {
		http.Error(w, "Unauthorized", http.StatusUnauthorized)
		return
	}

	tokenStr := strings.TrimPrefix(authHeader, "Bearer ")
	token, err := jwt.Parse(tokenStr, func(token *jwt.Token) (interface{}, error) {
		return &s.PrivateKey.PublicKey, nil
	})
	if err != nil || !token.Valid {
		http.Error(w, "Invalid Token", http.StatusUnauthorized)
		return
	}

	claims, ok := token.Claims.(jwt.MapClaims)
	if !ok {
		http.Error(w, "Invalid Claims", http.StatusBadRequest)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	_ = json.NewEncoder(w).Encode(claims)
}

func (s *Server) HandleSessionStream(w http.ResponseWriter, r *http.Request) {
	sessionID := r.URL.Query().Get("session_id")
	sess, exists := session.DefaultManager.GetSession(sessionID)
	if !exists {
		http.Error(w, "Session not found", http.StatusNotFound)
		return
	}

	w.Header().Set("Content-Type", "text/event-stream")
	w.Header().Set("Cache-Control", "no-cache")
	w.Header().Set("Connection", "keep-alive")

	flusher, ok := w.(http.Flusher)
	if !ok {
		http.Error(w, "Streaming unsupported", http.StatusInternalServerError)
		return
	}

	select {
	case <-sess.NotifyChan:
		authCode, _, _ := session.DefaultManager.IssueAuthCode(sessionID)
		redirectURL := fmt.Sprintf("%s?code=%s&state=%s", sess.RedirectURI, authCode, url.QueryEscape(sess.State))
		payload, _ := json.Marshal(map[string]string{
			"status":       "VERIFIED",
			"redirect_url": redirectURL,
		})
		_, _ = fmt.Fprintf(w, "data: %s\n\n", payload)
		flusher.Flush()
	case <-r.Context().Done():
		return
	case <-time.After(time.Until(sess.ExpiresAt)):
		_, _ = fmt.Fprintf(w, "data: {\"status\":\"EXPIRED\"}\n\n")
		flusher.Flush()
	}
}

import (
	"crypto/rand"
	"crypto/rsa"
	"crypto/x509"
	"encoding/base64"
	"encoding/json"
	"encoding/pem"
	"log"
	"math/big"
	"net/http"
	"os"

	"oneauth/internal/admin"
	"oneauth/internal/database"
	"oneauth/internal/oidc"
	"oneauth/internal/onebot"
)

func main() {
	dbPath := os.Getenv("ONEAUTH_DB_PATH")
	if dbPath == "" {
		dbPath = "oneauth.db"
	}
	port := os.Getenv("ONEAUTH_PORT")
	if port == "" {
		port = "8080"
	}
	issuer := os.Getenv("ONEAUTH_ISSUER")
	if issuer == "" {
		issuer = "http://localhost:" + port
	}

	_, err := database.InitDB(dbPath)
	if err != nil {
		log.Fatalf("初始化数据库失败: %v", err)
	}

	privKey, err := loadOrGenerateKey("jwt_rsa.key")
	if err != nil {
		log.Fatalf("初始化 RSA 密钥对失败: %v", err)
	}

	oidcSrv := &oidc.Server{
		IssuerURL:  issuer,
		PrivateKey: privKey,
		KeyID:      "oneauth-primary-key",
	}
	adminHandler := &admin.Handler{}

	mux := http.NewServeMux()

	// OIDC 协议端点
	mux.HandleFunc("GET /.well-known/openid-configuration", oidcSrv.HandleDiscovery)
	mux.HandleFunc("GET /.well-known/jwks.json", func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "application/json")
		_ = json.NewEncoder(w).Encode(map[string]interface{}{
			"keys": []map[string]interface{}{
				{
					"kty": "RSA",
					"use": "sig",
					"alg": "RS256",
					"kid": oidcSrv.KeyID,
					"n":   base64.RawURLEncoding.EncodeToString(privKey.N.Bytes()),
					"e":   base64.RawURLEncoding.EncodeToString(big.NewInt(int64(privKey.E)).Bytes()),
				},
			},
		})
	})
	mux.HandleFunc("GET /authorize", oidcSrv.HandleAuthorize)
	mux.HandleFunc("GET /login", oidcSrv.HandleLoginPage)
	mux.HandleFunc("GET /api/session/stream", oidcSrv.HandleSessionStream)
	mux.HandleFunc("POST /token", oidcSrv.HandleToken)
	mux.HandleFunc("GET /userinfo", oidcSrv.HandleUserInfo)

	// OneBot 反向 WebSocket
	mux.HandleFunc("/ws/onebot", onebot.HandleWebSocket)

	// 管理后台
	mux.HandleFunc("GET /admin", adminHandler.HandleAdminPage)
	mux.HandleFunc("POST /admin/api/settings", adminHandler.HandleSaveSettings)
	mux.HandleFunc("POST /admin/api/clients", adminHandler.HandleCreateClient)

	log.Printf("[OneAuth] 服务正在启动，监听端口: :%s\n", port)
	log.Printf("[OneAuth] OIDC Issuer: %s\n", issuer)
	if err := http.ListenAndServe(":"+port, mux); err != nil {
		log.Fatalf("HTTP 服务终止: %v", err)
	}
}

func loadOrGenerateKey(path string) (*rsa.PrivateKey, error) {
	data, err := os.ReadFile(path)
	if err == nil {
		block, _ := pem.Decode(data)
		return x509.ParsePKCS1PrivateKey(block.Bytes)
	}

	key, err := rsa.GenerateKey(rand.Reader, 2048)
	if err != nil {
		return nil, err
	}

	pemData := pem.EncodeToMemory(&pem.Block{
		Type:  "RSA PRIVATE KEY",
		Bytes: x509.MarshalPKCS1PrivateKey(key),
	})
	_ = os.WriteFile(path, pemData, 0600)
	return key, nil
}
